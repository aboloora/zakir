} return sheet; } function isRegistered_(email) { const values = usersSheet_().getDataRange().getValues(); return values.some(function (row) { return String(row[2]).trim().toLowerCase() === email; }); '' : value).trim().slice(0, max); } function esc_(s) { return String(s).replace(/[&<>"']/g, function (c) { return { '&': '&', '<': '<', '>': '>', '"': '"', "'": ''' }[c]; }); } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
